function saveMemberAssignedCodes(emailId) { const selects = document.querySelectorAll(`.member-code-select-${emailId}`); const codes = []; selects.forEach(sel => { const val = sel.value.trim().toUpperCase(); if (val && !codes.includes(val)) codes.push(val); }); // แปลงอีเมลเป็นตัวพิมพ์เล็กเสมอ เพื่อให้ตรงกับ Document ID ใน Firestore const docId = emailId.toLowerCase(); db.collection("user_profiles").doc(docId).set({ email: docId, assignedCodes: codes, assignedCode: codes.length > 0 ? codes[0] : "" }, { merge: true }).then(() => { showToast("บันทึกโค้ดส่วนลดอัตโนมัติสำเร็จ"); }).catch(err => { showToast("เกิดข้อผิดพลาดในการบันทึก: " + err.message); console.error(err); }); }
