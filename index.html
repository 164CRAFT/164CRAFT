function adjustModalQty(delta) { const input = document.getElementById("modalQtyInput"); let val = parseInt(input.value) || 1; if (currentSelectedProduct) { const scale = document.getElementById("detailScaleSelect").value; let maxLimit = 0; if (currentSelectedProduct.scalePrices && currentSelectedProduct.scalePrices[scale]) { const sp = currentSelectedProduct.scalePrices[scale]; if (typeof sp === 'object') { maxLimit = activeSaleMode === 'READY' ? (sp.readyStock || 0) : (sp.preorderUnlimited ? 999999 : (sp.preorderStock || 0)); } } else { maxLimit = activeSaleMode === 'READY' ? (currentSelectedProduct.readyStock || 0) : (currentSelectedProduct.preorderUnlimited ? 999999 : (currentSelectedProduct.preorderStock || 0)); } // กรณีสินค้าหมด (สต็อกเป็น 0) ให้จำนวนเป็น 0 เสมอ if (maxLimit <= 0) { input.value = 0; return; } val += delta; if (val > maxLimit) { val = maxLimit; showToast(`จำกัดจำนวนสูงสุดไม่เกินสต็อก (${maxLimit} ชิ้น)`); } } else { val += delta; } if (val < 1) val = 1; input.value = val; } function updateDetailStockStatus() { if (!currentSelectedProduct) return; const scale = document.getElementById("detailScaleSelect").value; const stockStatusEl = document.getElementById("detailStockStatus"); const qtyInput = document.getElementById("modalQtyInput"); if (!stockStatusEl) return; let isAct = false; let stockVal = 0; let isUnli = false; if (currentSelectedProduct.scalePrices && currentSelectedProduct.scalePrices[scale]) { const sp = currentSelectedProduct.scalePrices[scale]; if (activeSaleMode === "PREORDER") { isAct = sp.preorderActive; isUnli = sp.preorderUnlimited; stockVal = isUnli ? 999999 : (sp.preorderStock || 0); } else { isAct = sp.readyActive; stockVal = sp.readyStock || 0; } } else { if (activeSaleMode === "PREORDER") { isAct = currentSelectedProduct.preorderActive; isUnli = currentSelectedProduct.preorderUnlimited; stockVal = isUnli ? 999999 : (currentSelectedProduct.preorderStock || 0); } else { isAct = currentSelectedProduct.readyActive; stockVal = currentSelectedProduct.readyStock || 0; } } if (activeSaleMode === "PREORDER") { if (!isAct) { stockStatusEl.style.color = "#dc2626"; stockStatusEl.innerText = `❌ ปิดพรีออเดอร์`; if (qtyInput) qtyInput.value = 0; } else { stockStatusEl.style.color = "#c2410c"; stockStatusEl.innerText = `⏳ พรีออเดอร์ (${isUnli ? 'ไม่จำกัดจำนวน' : 'โควต้า ' + stockVal + ' ชิ้น'})`; if (qtyInput && parseInt(qtyInput.value) === 0) qtyInput.value = 1; } } else { if (!isAct || stockVal <= 0) { stockStatusEl.style.color = "#dc2626"; stockStatusEl.innerText = `❌ สินค้าหมด (Out of Stock)`; if (qtyInput) qtyInput.value = 0; } else { stockStatusEl.style.color = "#059669"; stockStatusEl.innerText = `📦 พร้อมส่ง (เหลือ ${stockVal} ชิ้น)`; if (qtyInput && parseInt(qtyInput.value) === 0) qtyInput.value = 1; } } }
